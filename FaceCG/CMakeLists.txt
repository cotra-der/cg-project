cmake_minimum_required(VERSION 3.20)
project(FaceCG LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FACECG_BUILD_APP "Build GLFW desktop application" ON)
option(FACECG_ENABLE_LIVE "Enable OpenCV and MediaPipe camera pipeline" OFF)
option(FACECG_BUILD_CAMERA_PROBE "Build standalone OpenCV camera proof" OFF)
add_library(facecg_core
  src/cg/DDA.cpp src/cg/Bresenham.cpp src/cg/Bezier.cpp
  src/cg/CatmullRom.cpp src/cg/ScanlineFill.cpp
  src/face/LandmarkMapper.cpp src/face/LandmarkSmoother.cpp src/face/FaceModel.cpp)
target_include_directories(facecg_core PUBLIC include)
include(CTest)
if(BUILD_TESTING)
  add_executable(facecg_tests tests/tests.cpp)
  target_link_libraries(facecg_tests PRIVATE facecg_core)
  add_test(NAME core COMMAND facecg_tests)
endif()
add_executable(facecg_preview tests/preview.cpp)
target_link_libraries(facecg_preview PRIVATE facecg_core)
if(FACECG_BUILD_CAMERA_PROBE)
  find_package(OpenCV REQUIRED COMPONENTS core videoio)
  add_executable(facecg_camera_probe tests/camera_probe.cpp)
  target_include_directories(facecg_camera_probe PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(facecg_camera_probe PRIVATE ${OpenCV_LIBS})
endif()
if(FACECG_BUILD_APP)
  find_package(OpenGL REQUIRED)
  find_package(glfw3 3.3 REQUIRED)
  add_executable(FaceCG src/main.cpp src/rendering/Renderer.cpp)
  target_link_libraries(FaceCG PRIVATE facecg_core glfw OpenGL::GL)
  if(FACECG_ENABLE_LIVE)
    find_package(OpenCV REQUIRED COMPONENTS core imgproc videoio)
    set(FACECG_MEDIAPIPE_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/mediapipe-c-sdk" CACHE PATH "Official MediaPipe C API header root")
    set(FACECG_MEDIAPIPE_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/mediapipe-native/mediapipe/tasks/c/libmediapipe.dll" CACHE FILEPATH "Native MediaPipe library (MinGW can link DLL directly)")
    if(NOT EXISTS "${FACECG_MEDIAPIPE_LIBRARY}" OR NOT EXISTS "${FACECG_MEDIAPIPE_INCLUDE}/mediapipe/tasks/c/vision/face_landmarker/face_landmarker.h")
      message(FATAL_ERROR "Native MediaPipe library and headers required; see docs/DEPENDENCIES.md")
    endif()
    add_library(MediaPipe::FaceLandmarker UNKNOWN IMPORTED)
    set_target_properties(MediaPipe::FaceLandmarker PROPERTIES IMPORTED_LOCATION "${FACECG_MEDIAPIPE_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${FACECG_MEDIAPIPE_INCLUDE}" INTERFACE_COMPILE_DEFINITIONS "MP_EXPORT=")
    target_sources(FaceCG PRIVATE src/face/LandmarkDetector.cpp)
    target_include_directories(FaceCG PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(FaceCG PRIVATE ${OpenCV_LIBS} MediaPipe::FaceLandmarker)
    target_compile_definitions(FaceCG PRIVATE FACECG_ENABLE_LIVE=1)
    add_executable(facecg_mediapipe_probe tests/mediapipe_probe.cpp)
    target_link_libraries(facecg_mediapipe_probe PRIVATE MediaPipe::FaceLandmarker)
    if(WIN32)
      # Official package filename differs from the DLL's embedded PE import name.
      foreach(target FaceCG facecg_mediapipe_probe)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FACECG_MEDIAPIPE_LIBRARY}" "$<TARGET_FILE_DIR:${target}>/mediapipe_source.dll")
      endforeach()
    endif()
  endif()
endif()
